package tally

import (
	"reflect"
	"testing"
)

func TestWordsSplitsOnPunctuation(t *testing.T) {
	got := Words("Hello, world! hello again.")
	want := []string{"Hello", "world", "hello", "again"}
	if !reflect.DeepEqual(got, want) {
		t.Fatalf("Words = %q, want %q", got, want)
	}
}

func TestCountCountsEachWord(t *testing.T) {
	got := Count("a b a")
	if got["a"] != 2 || got["b"] != 1 || len(got) != 2 {
		t.Fatalf("Count = %v", got)
	}
}

func TestLinesCountsNewlineEndedLines(t *testing.T) {
	if n := Lines("one\ntwo\n"); n != 2 {
		t.Fatalf("Lines = %d, want 2", n)
	}
}
